* Insert date

From Aaron

#+begin_src emacs-lisp :tangle no
(defun c/insert-log-date ()
  "Insert a date in my typical log format.

e.g. Fri Jan 07 2022"
  (interactive)
  (insert (format-time-string "%a %b %-d %Y")))
#+end_src

* Comment attribution

From Aaron

#+begin_src emacs-lisp :tangle no
(defun c/insert-comment-attribution ()
  "Insert a comment attribution.

e.g. - Aaron, Fri Jan 07 2022"
  (interactive)
  (insert "- Aaron")
  (save-excursion
    (insert ", ")
    (c/insert-log-date)))

(define-key global-map (kbd "C-c !") #'c/insert-log-date)
(define-key global-map (kbd "C-c C--") #'c/insert-comment-attribution)
#+end_src
