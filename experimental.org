* Running tests

You can call ==vterm-send-string== when the current buffer is in ==vterm-mode==, which if you include a "\n" it will "hit enter" as if you were typing commands

#+begin_src emacs-lisp :tangle no

  (defun custom/test-component-project ()
    (interactive)
    (if (equal current-prefix-arg nil)
        (custom/split-with-process "ct")
        (custom/split-with-process "s && ct")
      ))

  (defun custom/test-web-project ()
    (interactive)
    (if (equal current-prefix-arg nil)
        (custom/split-with-process "rs")
        (custom/split-with-process "s && rs")
      ))

  (defun custom/split-with-process (command)
    (tab-bar-new-tab)
    (vterm t)
    (vterm-send-string command)
    (vterm-send-return)
    (split-window-right)
    (vterm t)
    )

#+End_src

* Comment attribution

From Aaron

#+begin_src emacs-lisp :tangle no

(defun c/insert-comment-attribution ()
  "Insert a comment attribution.

e.g. - Aaron, Fri Jan 07 2022"
  (interactive)
  (insert "- Aaron")
  (save-excursion
    (insert ", ")
    (c/insert-log-date)))

(define-key global-map (kbd "C-c !") #'c/insert-log-date)
(define-key global-map (kbd "C-c C--") #'c/insert-comment-attribution)

* Timing
Aaron provided his code.

In early init:

#+begin_src emacs-lisp :tangle no
(defvar c/early-init-el-start-time (current-time) "Time when early-init.el was started")
(defvar c/section-split-time c/early-init-el-start-time)

(defun c/get-section-split-time ()
  "Return the time since the init or since the last time this was called."
  (let ((now (current-time)))
    (prog1
        (float-time (time-subtract now c/section-split-time))
      (setq c/section-split-time now))))

(defvar c/previous-section nil)

(defun c/start-section (file heading)
  "Print the split time for the previous heading."
  (when (and c/previous-section
             (not noninteractive))
    (let ((inhibit-message t)
          (time (* 1000 (c/get-section-split-time))))
      (when (> time 3)
        (message "%6s • %4dms • %s" (car c/previous-section) time (cdr c/previous-section)))))
  (setq c/previous-section (and file heading (cons file heading))))
#+end_src

#+begin_src emacs-lisp :tangle no
;;; init.el -*- lexical-binding: t; -*-

(c/start-section "init" "Start")

(defvar c/init-el-start-time (current-time) "Time when init.el was started")

(c/start-section "init" "elpaca")

;; ELPACA SETUP

(elpaca (literate-config :host github :repo "aaronjensen/emacs-literate-config" :protocol ssh :wait t))

(add-hook 'literate-config-before-section-hook
          (lambda (section)
            (c/start-section "config" section)))

(unless noninteractive
  (add-hook 'elpaca-after-init-hook
            (lambda ()
              (let ((inhibit-message t))
                (message "early-init.el: %d ms, init.el: %d ms, config.el: %d ms, post: %d ms\nInitial: %d ms, Total: %d ms"
                         (* 1000 (float-time (time-subtract c/init-el-start-time c/early-init-el-start-time)))
                         (* 1000 (float-time (time-subtract c/config-el-start-time c/init-el-start-time)))
                         (* 1000 (float-time (time-subtract c/config-el-end-time c/config-el-start-time)))
                         (* 1000 (float-time (time-subtract (current-time) c/config-el-end-time)))
                         (* 1000 (float-time (time-subtract c/config-el-end-time c/early-init-el-start-time)))
                         (* 1000 (float-time (time-subtract (current-time) c/early-init-el-start-time)))))) -90))

(defvar c/config-el-start-time (current-time))

(c/start-section "init" "Load Config")
(literate-config-init)
(defvar c/config-el-end-time (current-time))
